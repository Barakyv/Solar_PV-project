# Source of the probe:
"""Does gpu_peak_mb (torch.cuda.max_memory_allocated) move with Python's garbage-collector timing?

Same model, same data, same seed; only the number of Python container objects alive before the
fit differs (that shifts when the cyclic collector runs). Weights are compared to prove the
computation is the same.
"""
import gc
import os
import sys

os.environ["CUBLAS_WORKSPACE_CONFIG"] = ":4096:8"
sys.path[:0] = [r"C:\Users\204323141\Desktop\solar\src", r"C:\Users\204323141\Desktop\solar\tests\gpu"]
import test_checkpoint_gpu as T  # noqa: E402
import torch  # noqa: E402
from test_neural_gpu import TFT, scope_arrays  # noqa: E402

from solarpv.models import neural_torch, tft_model  # noqa: E402

arrays = {**scope_arrays("train", 400, 0, short=100), **scope_arrays("val", 100, 1, short=True)}
params = {**TFT, "lstm_layers": 2, "hidden_size": 64}


def run(k, collect_first):
    junk = [[] for _ in range(k)]  # tracked containers: shift the allocation counters of the gc
    if collect_first:
        gc.collect()
    torch.cuda.empty_cache()
    model, res, _ = T.fit_epochs(tft_model, params, arrays, epochs=3)
    peak = res.gpu_peak_mb
    w = {name: p.detach().cpu().clone() for name, p in model.state_dict().items()}
    del junk
    return peak, w


base_peak, base_w = run(0, True)
print(f"k=0 (reference)   peak={base_peak!r}")
for k in (0, 1, 3, 17, 101, 700, 5000):
    peak, w = run(k, False)
    same_w = all(torch.equal(w[n], base_w[n]) for n in w)
    print(f"k={k:<5} peak={peak!r:<22} diff vs reference={peak - base_peak:+.6f} MB  weights equal={same_w}")

gc.disable()
peak, w = run(0, True)
print(f"gc disabled       peak={peak!r:<22} diff vs reference={peak - base_peak:+.6f} MB  "
      f"weights equal={all(torch.equal(w[n], base_w[n]) for n in w)}")
gc.enable()
